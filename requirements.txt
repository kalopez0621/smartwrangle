# ============================================================
# SmartWrangle: Python Requirements
# ============================================================
# This file tells your laptop (through run.bat) and Streamlit
# Community Cloud which packages to install.
#
# VERSION PINNING:
#   Every package is pinned to the exact version (==) that was
#   tested on Python 3.12. This makes the laptop and the live
#   app run the same code. To upgrade a package later, change
#   its number here, run run.bat, and test the app again.
#
#   pandas 3.x needs Python 3.11 or newer, so the Streamlit
#   Cloud app must be set to Python 3.12 as well.
# ============================================================


# ── Core app framework ──────────────────────────────────────
# Streamlit powers the entire UI, tabs, widgets, session state,
# and the file upload system.
streamlit==1.65.0


# ── Data manipulation ────────────────────────────────────────
# pandas loads, cleans, and transforms the data.
pandas==3.0.6

# numpy handles numeric operations (IQR, skew, log transform).
numpy==2.5.4

# openpyxl is required for pd.read_excel() to open .xlsx files.
openpyxl==3.1.5


# ── Visualization ────────────────────────────────────────────
# plotly draws every chart in the Discover tab.
plotly==7.1.0


# ── Machine learning / preprocessing ─────────────────────────
# scikit-learn provides StandardScaler and MinMaxScaler,
# used by the "Preparation for Analysis" tools.
scikit-learn==1.9.1


# ── AI Analyst ────────────────────────────────────────────────
# anthropic is the official Python SDK for the Claude API.
anthropic==1.13.0

# python-dotenv reads ANTHROPIC_API_KEY from a local .env file
# on your laptop. On Streamlit Cloud the key comes from the
# app's Secrets settings instead, so this is only used locally.
python-dotenv==1.2.4


# ── Arrow serialization ───────────────────────────────────────
# pyarrow is how Streamlit sends tables to the browser.
# Pinned so it always matches the pandas version above.
pyarrow==25.0.1
